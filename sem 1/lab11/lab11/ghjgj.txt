#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    ifstream fin("input.txt");   // открываем файл для чтения
    if (!fin) {
        cerr << "Не удалось открыть файл\n";
        return 1;
    }

    int N, M;
    fin >> N >> M;
    vector<string> a(N);
    for (int i = 0; i < N; ++i) {
        a[i].clear();
        for (int j = 0; j < M; ++j) {
            char c;
            fin >> c;            // читаем символ (пропуская пробелы)
            a[i].push_back(c);
        }
    }

    int sx=-1, sy=-1, ex=-1, ey=-1;
    for (int i=0;i<N;i++) for (int j=0;j<M;j++){
        if(a[i][j]=='S'){ sx=i; sy=j; }
        if(a[i][j]=='E'){ ex=i; ey=j; }
    }

    const int INF = 1e9;
    vector<vector<int>> dist(N, vector<int>(M, INF));

    // простая очередь на массивах
    int maxSize = N*M;
    vector<int> qx(maxSize), qy(maxSize);
    int head=0, tail=0;

    int dx[8] = {-1,-1,-1,0,0,1,1,1};
    int dy[8] = {-1,0,1,-1,1,-1,0,1};

    dist[sx][sy]=0;
    qx[tail]=sx; qy[tail]=sy; ++tail;

    while(head<tail){
        int x=qx[head], y=qy[head]; ++head;
        if(x==ex && y==ey) break;
        for(int k=0;k<8;k++){
            int nx=x+dx[k], ny=y+dy[k];
            if(nx<0nx>=Nny<0||ny>=M) continue;
            if(a[nx][ny]=='#') continue;
            if(dist[nx][ny]>dist[x][y]+1){
                dist[nx][ny]=dist[x][y]+1;
                qx[tail]=nx; qy[tail]=ny; ++tail;
            }
        }
    }

    if(dist[ex][ey]==INF) cout<<-1<<"\n";
    else cout<<dist[ex][ey]<<"\n";
}

void QuickSort(int a, int b, int* arr) {
    if(a>=b)
        return;
    int k = (rand()*rand()) % (b-a+1) + a;
    int m = arr[k];
    int l = a - 1;
    int r = b + 1;
    while(true) {
        do l++; while(arr[l] < m);
        do r--; while(arr[r] > m);
        if(l >= r) 
            break;
        std::swap(arr[l], arr[r]);
    }
    l = r;
    r++;
    QuickSort(a, l, arr);
    QuickSort(r, b, arr);
}

size_t fact(size_t n, size_t* factorials) {
    if (factorials[n] != 0)
        return factorials[n];
    factorials[n] = fact(n-1, factorials)*n; 
    return factorials[n];
}
